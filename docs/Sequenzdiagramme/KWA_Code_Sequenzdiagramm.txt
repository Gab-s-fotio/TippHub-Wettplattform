@startuml
title Zielgruppe_Festlegen

actor ":Kommerzielle Wettanbieter" as KWA
participant ":System" as System


KWA -> System: "Zielgruppen-Ankündigung" wählen
activate System

System --> KWA: Targeting-Optionen anzeigen\n(Geofilter, Interessen, Aktivitätslevel)
activate KWA
KWA -> System: Zielgruppe konfigurieren\nund Link festlegen

loop Unzureichende Abdeckung
    System -> KWA: Warnung: Geringe Reichweite\n(geschätzte Reichweite < Minimum)
  

System -> System: Digitale Signatur generieren
System -> System: Ankündigung validieren
 
     System --> KWA: Erfolgsmeldung anzeigen\n"Zielgruppe gespeichert"
    System --> KWA: Verteilung bestätigen\n(Push, Community-Feeds)
deactivate KWA
loop Fehler
    System --> KWA: Fehlermeldung anzeigen\n"Konfiguration ungültig"
end

deactivate System
@enduml

        KWA -> System: Weiterleitung zur Guthabenaufladung
    end

        System --> KWA: Option zur manuellen THB-Freigabe

participant ":System" as System


KWA -> System: Klick auf gesponserte Ankündigung

    System -> System: Tracking-Pixel triggern
    System --> KWA: Öffnet Zielseite in neuem Tab
    activate KWA

@startuml
title THMCoins an PUS gutschreiben

actor ":Kommerzielle Wettanbieter" as KWA
participant ":System" as System


KWA -> System: Öffnet THMCoin-Verwaltung
activate System
System --> KWA: Eingabemaske anzeigen\n(PUS-Suche, Betrag, Verwendungszweck)
activate KWA
KWA -> System: PUS auswählen und Betrag eingeben
System -> System: Validierung durchführen\n(PUS existiert? Guthaben ausreichend? Limit eingehalten?)


    System --> KWA: Bestätigungsdialog anzeigen\n(Betrag, Empfänger, Gebühren)
    KWA -> System: Transaktion bestätigen
    System -> System: Coin-Transfer ausführen\n(Saldo aktualisieren, Transaktionsprotokoll)
    System --> KWA: Erfolgsmeldung\n"Transfer erfolgreich"
    System --> KWA: Transaktionsbeleg senden
deactivate KWA
loop Validierungsfehler
    System --> KWA: Fehlermeldung anzeigen\n(z.B. "Tageslimit überschritten")
end

deactivate System
@enduml
